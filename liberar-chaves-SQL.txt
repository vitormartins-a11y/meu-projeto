-- Libera as chaves guardadas que ficaram marcadas "sem cota hoje" por engano
-- (um modelo do Gemini sem cota deixava a chave inteira parada até as 04:00).
-- Rode DEPOIS de publicar a função "ia" nova. Se alguma chave estiver mesmo sem cota, ela volta a descansar sozinha.
update chaves_ia set descanso_ate = null where descanso_ate > now();
