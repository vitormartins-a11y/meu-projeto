-- Onde foram os 8 mil pedidos de flashcards e os 6 mil de resumo das últimas 48 horas: os temas que mais pediram, quantas
-- vezes cada um e o estado de cada um (se a gravação está pesada demais, o tema pode estar refazendo a mesma coisa).
-- SÓ LÊ, não muda nada e não mostra nenhuma chave. Rode e me mande o resultado inteiro. (versão 2, mais leve)
with
u as (select u.materia_id, coalesce(u.tarefa, 'apostila') as tarefa, u.ok, u.em
        from uso_ia u where u.em > now() - interval '48 hours' and u.uso is distinct from 'estudo' and u.materia_id is not null),
conta as (select materia_id, count(*) as n,
                 count(*) filter (where tarefa = 'flashcards' and ok) as fc_ok, count(*) filter (where tarefa = 'flashcards') as fc,
                 count(*) filter (where tarefa = 'resumo' and ok) as rs_ok, count(*) filter (where tarefa = 'resumo') as rs,
                 count(*) filter (where tarefa = 'questoes' and ok) as q_ok, count(*) filter (where tarefa = 'questoes') as q,
                 count(*) filter (where tarefa = 'apostila' and ok) as ap_ok, count(*) filter (where tarefa = 'apostila') as ap,
                 count(distinct date_trunc('hour', em)) filter (where tarefa = 'flashcards' and ok) as horas_fc
            from u group by materia_id order by count(*) desc limit 15),
estado as (select r.materia_id, pg_column_size(r.dados) as tam, r.dados->>'modelo' as modelo, r.dados->>'extras' as extras,
                  r.dados->>'qversao' as qversao, r.dados->'rascunho'->>'etapa' as etapa, left(r.dados->>'geradoEm', 16) as publicado
             from resumos r where r.materia_id in (select materia_id from conta))
select m.materia as tema, c.n as pedidos,
       c.fc_ok || ' de ' || c.fc as flashcards, c.rs_ok || ' de ' || c.rs as resumo, c.q_ok || ' de ' || c.q as questoes, c.ap_ok || ' de ' || c.ap as apostila,
       c.horas_fc as horas_com_flashcards, pg_size_pretty(e.tam::bigint) as tamanho_do_tema, coalesce(e.modelo, 'antigo') as modelo,
       coalesce(e.extras, '-') as extras, coalesce(e.qversao, '-') as qversao, coalesce(e.etapa, '-') as etapa_do_rascunho, coalesce(e.publicado, '-') as publicado_em
  from conta c join materias m on m.id = c.materia_id left join estado e on e.materia_id = c.materia_id
 order by c.n desc;
