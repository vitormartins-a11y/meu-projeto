      user_id: usuario, chave_id: prov === 'gemini' && propria ? chaveId : null, materia_id: typeof materia_id === 'string' ? materia_id : null,
      pediu: usuario, turma_id: typeof turma_id === 'string' && /^[0-9a-f-]{36}$/i.test(turma_id) ? turma_id : null, ...(modo === 'estudo' ? { uso: 'estudo' } : {}), ...extra });
    // ferramentas de estudo: o Gemini da reserva primeiro, depois o Groq e, por último, a chave da turma
    // (se o app já pediu para pular o Gemini, porque ele demorou, vai direto para o Groq)
    const estudoPrimeiro = podeGuardadas && modo === 'estudo' && !provedor && !(Array.isArray(evitar) ? evitar as string[] : []).includes('gemini');
    const ordemFinal: Provedor[] = estudoPrimeiro ? ['gemini', ...fila.filter(x => x !== 'gemini'), ...(fila.includes('gemini') ? ['gemini' as Provedor] : [])] : fila;
    let guardadasTentadas = false, groqGuardadasTentadas = false, geminiLento = false, pulouPaga = false;
    for (const [i, prov] of ordemFinal.entries()) {
      if (prazo - Date.now() < 15_000) break;
      if (prov === 'gemini' && geminiLento) continue;   // o Gemini já demorou demais neste pedido: só as outras IAs
      ultimo = prov;
      if (prov === 'gemini' && podeGuardadas && !guardadasTentadas) {   // primeiro o time de chaves guardadas dos alunos
        guardadasTentadas = true;
        // o Gemini das chaves guardadas tem um tempo limite, para sobrar tempo para o Groq se ele demorar
        const prazoG = Math.min(prazo, Date.now() + (tarefaOk === 'curta' ? 35_000 : estudoPrimeiro ? 70_000 : 85_000));
        const g = await comChavesGuardadas(p, prazoG, falhas, anotar, modo);
        if (g === 'lento') { geminiLento = true; continue; }
        if (g) return g;
        if (prazo - Date.now() < 15_000) break;
        if (estudoPrimeiro) continue;                     // reserva sem cota: vai para o Groq antes da chave da turma
      }
      // o Groq: primeiro o time de chaves que os alunos deixaram (SQL v51), depois a chave da turma
      if (prov === 'groq' && podeGuardadas && !groqGuardadasTentadas && !p.minimo) {
        groqGuardadasTentadas = true;
        const g = await comChavesGroq(p, prazo, falhas, anotar, tarefaOk);
        if (g) return g;
        if (prazo - Date.now() < 15_000) break;
        if (!chaveDe('groq')) continue;
      }
      // pedido "só grátis" (refazer o acervo no modelo novo): as chaves pagas ficam de fora
      if (so_gratis && !propria && ehPaga(prov)) { pulouPaga = true; continue; }
      // com modelo mínimo do Gemini, as outras IAs não entram (a apostila espera a cota do Gemini)
      if (p.minimo && prov !== 'gemini') { pulouPaga = true; continue; }
      if (!chaveDe(prov)) continue;                    // (o Groq entrou só pelas chaves dos alunos, e elas não deram)
      const key = chaveDe(prov)!;
      // se ainda há outra IA na fila, o Gemini tem um tempo limite: travado, ele não gasta o prazo inteiro do pedido
      const outraDepois = prov === 'gemini' && ordemFinal.slice(i + 1).some(x => x !== 'gemini');
      const r = prov === 'gemini' ? await chamarGemini(key, p, outraDepois ? Math.min(prazo, Date.now() + 85_000) : prazo, rastro) : await chamarChat(prov, key, p, prazo, rastro, tarefaOk);
      if (r.ok) { await anotar(true, 200, prov, r.modelo, r.uso); return json({ texto: r.texto, modelo: r.modelo, provedor: prov, fim: r.fim, uso: r.uso, tentativas: falhas, ...(propriaRecusada ? { propriaRecusada } : {}), ...(p.imagens ? { visao: true } : {}) }); }
      if (prov === 'gemini' && propria && r.chave) propriaRecusada = true;
      motivos.push(r);
      if (r.cota && !descansoAte[prov] && !(prov === 'gemini' && propria)) descansoAte[prov] = Date.now() + 60_000;
      if (r.lento && outraDepois) { motivos.pop(); geminiLento = true; continue; }   // Gemini travado: segue para a próxima IA
      if (r.lento) { await anotar(false, 504, prov, null, null); return json({ error: r.msg, lento: prov, tentativas: falhas }, 504); }   // o app repete o pedido sem esta IA
    }
    // só grátis e todas as chaves grátis sem cota (ou nenhuma livre): é pedido de cota (a fila espera e tenta depois)
    // (de que limite foi: a cota do dia acabou, ou foram pedidos demais no mesmo minuto; o app espera conforme o caso)
    const limite = falhas.some(x => /429 minuto$/.test(x)) ? 'minuto' : falhas.some(x => /429 dia$/.test(x)) ? 'dia' : null;
    if (pulouPaga && !motivos.length && !geminiLento) { await anotar(false, 429, ultimo, null, null); return json({ error: 'As chaves grátis estão sem cota agora.', cota: true, soGratis: true, limite, tentativas: falhas }, 429); }
    // o Gemini demorou e não sobrou tempo para outra IA: o app repete o pedido já sem o Gemini
    if (geminiLento && !motivos.length) { await anotar(false, 504, 'gemini', null, null); return json({ error: 'O Gemini demorou demais.', lento: 'gemini', tentativas: falhas }, 504); }
    // a chave própria foi recusada ou está sem cota: o app avisa a pessoa (e, fora da ajuda à fila, tenta de novo com a da turma)
    const falhouPropria = !!propria && ultimo === 'gemini' && fila.length === 1;
    if (motivos.length === 1 && motivos[0].chave) { await anotar(false, 502, ultimo, null, null); return json({ error: falhouPropria ? 'A sua chave do Gemini foi recusada. Confira se copiou a chave inteira.' : motivos[0].msg, propria: falhouPropria, chave: true, tentativas: falhas }, 502); }
    // "texto grande demais" para uma IA não conta: se as outras estão sem cota, o pedido é de cota (o robô faz pausa)
    const cota = motivos.some(m => m.cota) && motivos.every(m => m.cota || m.chave || m.status === 413);
    await anotar(false, cota ? 429 : 503, ultimo, null, null);
    const nomes = fila.map(x => NOME[x]).join(' e ');
    return json({
      error: cota ? `Limite gratuito atingido por agora (${nomes}).` : (motivos.at(-1)?.msg || 'As IAs estão sobrecarregadas agora.'),
      cota, limite, propria: falhouPropria || (!!propria && motivos[0]?.cota && fila[0] === 'gemini'), tentativas: falhas,
    }, cota ? 429 : 503);
  } catch (e) {
    console.error(e);                                   // o detalhe fica no registro da função; a pessoa vê uma mensagem simples
    return json({ error: 'Erro no servidor da IA. Tente de novo em instantes.' }, 500);
  }
});
